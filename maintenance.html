(() => { "use strict"; let redirecting = false; function goToMaintenance(reason) { if (redirecting) return; redirecting = true; console.error( "[Crystal Kingdom] Maintenance mode:", reason ); const url = new URL( "./maintenance.html", window.location.href ); url.searchParams.set("reason", "game-error"); window.location.replace(url.href); } // Catch errors that come specifically from game.js. window.addEventListener("error", (event) => { const source = String(event.filename || ""); const isGameError = /\/game\.js(?:[?#]|$)/i.test(source); const isGameLoadError = event.target instanceof HTMLScriptElement && /\/game\.js(?:[?#]|$)/i.test(event.target.src); if (isGameError || isGameLoadError) { goToMaintenance({ message: event.message || "Game script failed", file: source || event.target?.src || "game.js", line: event.lineno || 0 }); } else { // Image, music, and unrelated resource errors should not // automatically take the entire game offline. console.warn("[Crystal Kingdom] Resource error:", event); } }, true); // Log promise errors without automatically redirecting. // Browser autoplay and voice features can reject promises // without the entire game being broken. window.addEventListener("unhandledrejection", (event) => { console.error( "[Crystal Kingdom] Unhandled promise rejection:", event.reason ); }); // If initialization never succeeds, switch to maintenance mode. window.addEventListener("load", () => { window.setTimeout(() => { if (!window.gemMinerReady) { goToMaintenance( "The game did not finish initializing." ); } }, 5000); }); })();
